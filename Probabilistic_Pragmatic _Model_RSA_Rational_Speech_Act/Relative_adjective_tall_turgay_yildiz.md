var data = {
  
  "human"   :   human, 
  "tree"    :   tree

//---------------------------------------------------------------------------------

var prior_height = function(item) {
  
  return Infer({method: "enumerate"}, function(){

  var uniformInt    =   sample(RandomInteger({n: 100}))
  var height        =   data[item].heights[uniformInt]; 
  
  return height
  });
};

//---------------------------------------------------------------------------------
var utterancePrior_1 = function() {
  
  var uttProbs = map(function(u) {return Math.exp(-cost_1[u]) }, utterances_1);
  
  return categorical(uttProbs, utterances_1);
};
//---------------------------------------------------------------------------------
var utterancePrior_2 = function() {
  
  var uttProbs = map(function(u) {return Math.exp(-cost_2[u]) }, utterances_2);
  
  return categorical(uttProbs, utterances_2);
};
//---------------------------------------------------------------------------------
// alpha = 2.5 and beta = 2.5 for human heights
// alpha = 10  and beta = 2.5 for tree  heights

var  alpha  =  10 
var  beta   =  2.5
//---------------------------------------------------------------------------------
var cost_1 = {
             "tall" : 1,
              ""    : 0
};
//---------------------------------------------------------------------------------
var cost_2 = {
             "yes"    : 1,
              ""      : 0
};
//---------------------------------------------------------------------------------
var utterances_1    = ["tall", ""];
var utterances_2    = ["yes", ""];
//---------------------------------------------------------------------------------
var meaning_1 = function(utterance, aggr) {
  
  if (utterance == "tall") {
    return flip(aggr); 
  } else {
    return true;
  }
};
//---------------------------------------------------------------------------------
var meaning_2 = function(utterance, aggr) {
  
  if (utterance == "yes") {
    return flip(aggr * (1 - aggr)) ; 
  } else {
    return true;
  }
};
//---------------------------------------------------------------------------------
var literal_listener_1 = cache(function(utterance, item) {
  return Infer({method: "enumerate"}, function() {
// this prior is same as prior_height. However, since indices must be the same,
// it is defined here. 
    var uniformInt  =   sample(RandomInteger({n: 100}))
    var height      =   data[item].heights[uniformInt]; 
    var aggr        =   data[item].probabilities[uniformInt]; 

    condition(meaning_1(utterance, aggr))

    return height ;
  });
});
//---------------------------------------------------------------------------------
var literal_listener_2 = cache(function(utterance, item) {
  return Infer({method: "enumerate"}, function() {
    
    var uniformInt  =   sample(RandomInteger({n: 100}))
    var threshold   =   data[item].heights[uniformInt]; 
    var aggr        =   data[item].probabilities[uniformInt]; 

    condition(meaning_2(utterance, aggr))

    return threshold ;
  });
});
//---------------------------------------------------------------------------------
//---------------------------------------------------------------------------------
var speaker_1 = cache(function(height, item) {
  
  return Infer({method: "enumerate"}, function() { 

    var utterance = utterancePrior_1();
    
    factor( alpha * literal_listener_1(utterance, item).score(height) );

    return utterance;
  })    
});
//---------------------------------------------------------------------------------
var speaker_2 = cache(function(threshold, item) {

  return Infer({method: "enumerate"}, function() { 

    var utterance = utterancePrior_2();
    
    factor( beta * literal_listener_2(utterance, item).score(threshold) );

    return utterance;
  });
});
//---------------------------------------------------------------------------------
var pragmaticListener = function(q, utterance, item) {
  
  return Infer({method: "enumerate"}, 
  function() {
    
    var height_d  =  prior_height(item)
    var height    =  sample(height_d)
    if (q == "speaker_1"){
    factor( speaker_1(height, item).score(utterance) );
    }
    else{factor( speaker_2(height, item).score(utterance) ); }
    
    return height;
  });
};
//---------------------------------------------------------------------------------

//---------------------------------------------------------------------------------
var marginalize = function(dist, key){
  
  return Infer({method: "enumerate"}, function(){
    
    return sample(dist)[key];
  })
}
