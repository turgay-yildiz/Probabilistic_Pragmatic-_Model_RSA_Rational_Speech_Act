var data = {
  
  "bucket"  :   bucket, 
  "pool"    :   pool,

};
//---------------------------------------------------------------------------------

var prior_aol = function(item) {
  
  return Infer({method: "enumerate"}, function(){

  var uniformInt    =   sample(RandomInteger({n: 100}))
  var aol           =   data[item].AoL[uniformInt]; 
  
  return aol
  });
};

//---------------------------------------------------------------------------------
var utterance_prior_1 = function() {
  
  var uttProbs = map(function(u) {return Math.exp(-cost_1[u]) }, utterances_1);
  
  return categorical(uttProbs, utterances_1);
};

var utterance_prior_2 = function() {
  
  var uttProbs = map(function(u) {return Math.exp(-cost_2[u]) }, utterances_2);
  
  return categorical(uttProbs, utterances_2);
};
//---------------------------------------------------------------------------------
// Alpha is 2.5 both for bucket and pool
// Beta  is also 2.5 for bucket and pool

var  alpha  =  2.5
var  beta   =  2.5 
//---------------------------------------------------------------------------------
var cost_1 = {
             "empty" : 1,
              ""     : 0
};
var cost_2 = {
             "yes"    : 1,
              ""      : 0
};
//---------------------------------------------------------------------------------
var utterances_1  = ["empty", ""];
var utterances_2  = ["yes", ""];
//---------------------------------------------------------------------------------
var meaning_1 = function(utterance, aggr) {
  
  if (utterance == "empty") {
    return flip(aggr); 
  } else {
    return true;
  }
};
//---------------------------------------------------------------------------------
var meaning_2 = function(utterance, aggr) {
  
  if (utterance == "yes") {
    return flip(aggr * (1 - aggr)) ; 
  } else {
    return true;
  }
};
//---------------------------------------------------------------------------------
var literal_listener_1 = cache(function(utterance, item) {
  return Infer({method: "enumerate"}, function() {
// this prior is same as prior_aol. However, since indices must be the same,
// it is defined here. 
    var uniformInt  =   sample(RandomInteger({n: 100}))
    var aol         =   data[item].AoL[uniformInt]; 
    var aggr        =   data[item].probabilities[uniformInt]; 

    condition(meaning_1(utterance, aggr))

    return aol ;
  });
});

var literal_listener_2 = cache(function(utterance, item) {
  return Infer({method: "enumerate"}, function() {
    
    var uniformInt  =   sample(RandomInteger({n: 100}))
    var threshold   =   data[item].AoL[uniformInt]; 
    var aggr        =   data[item].probabilities[uniformInt]; 

    condition(meaning_2(utterance, aggr))

    return threshold ;
  });
});
//---------------------------------------------------------------------------------
var speaker_1 = cache(function(aol, item) {

  return Infer({method: "enumerate"}, function() { 

    var utterance = utterance_prior_1();
    
    factor( alpha * literal_listener_1(utterance, item).score(aol) );

    return utterance;
  });
});
//---------------------------------------------------------------------------------
var speaker_2 = cache(function(threshold, item) {

  return Infer({method: "enumerate"}, function() { 

    var utterance = utterance_prior_2();
    
    factor( beta * literal_listener_2(utterance, item).score(threshold) );

    return utterance;
  });
});
//---------------------------------------------------------------------------------
var pragmaticListener = function(q, utterance, item) {
  
  return Infer({method: "enumerate"}, 
  function() {
    
    var aol_d     =  prior_aol(item)
    var aol       =  sample(aol_d)
    
    if (q == "speaker_1"){
    factor( speaker_1(aol, item).score(utterance) )
    }
    else{ factor(speaker_2(aol, item).score(utterance)) }
    
    return aol;
  });
};
//---------------------------------------------------------------------------------
var marginalize = function(dist, key){
  
  return Infer({method: "enumerate"}, function(){
    
    return sample(dist)[key];
  })
}
